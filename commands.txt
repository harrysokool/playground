**Added (1)**

*How many active Agency agents have both a currently effective typed licence record and an education record?*
```sql
-- Separate EXISTS checks avoid multiplying agents by licence and education rows.
-- Currently effective = date validity only; not proof of regulatory authorization.
SELECT
  COUNT(DISTINCT `m`.`INTAGTSEQ`) AS `active_agents_with_effective_licence_and_education`
FROM `aiahk_dna_p_catalog`.`dna_gia_base`.`gold_agent_master` AS `m`
WHERE `m`.`row_type` = 'Agency'
  AND `m`.`agent_status_code` IN ('00', '10')
  AND EXISTS (
    SELECT 1
    FROM `aiahk_dna_p_catalog`.`dna_gia_base`.`gold_agent_license` AS `l`
    WHERE `l`.`INTAGTSEQ` = `m`.`INTAGTSEQ`
      AND `l`.`row_type` = `m`.`row_type`
      AND `l`.`license_type` IS NOT NULL
      AND `l`.`license_eff_date` IS NOT NULL
      AND `l`.`license_exp_date` IS NOT NULL
      AND `l`.`license_eff_date` <= CURRENT_DATE()
      AND `l`.`license_exp_date` >= CURRENT_DATE()
  )
  AND EXISTS (
    SELECT 1
    FROM `aiahk_dna_p_catalog`.`dna_gia_base`.`gold_agent_education` AS `e`
    WHERE `e`.`INTAGTSEQ` = `m`.`INTAGTSEQ`
  )
```
