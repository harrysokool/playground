for key in ("MSI_ENDPOINT", "MSI_SECRET",

import json, os

        c = json.load(f)
    t = c.get("tools", {})
    print("Profile:", t.get("profile", "Not set"))
    print("Denied tools:", t.get("deny", []))
    print("Sandbox:", c.get("agents", {}).get("defaults", {}).get("sandbox", {}).get("mode", "Not set"))
