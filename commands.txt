import json
from pathlib import Path

home = Path.home()
proxy_env = home / "cloudfiles/code/Users/Harry-CC.Wang/playground/riley_proxy/.env"
config_path = home / ".openclaw/openclaw.json"

values = {}
for line in proxy_env.read_text().splitlines():
    if "=" in line and not line.lstrip().startswith("#"):
        key, value = line.split("=", 1)
        values[key.strip()] = value.strip().strip('"').strip("'")

deployment = values["AZURE_DEPLOYMENT"]
config = json.loads(config_path.read_text())

backup = config_path.with_suffix(".json.riley-backup")
backup.write_text(json.dumps(config, indent=2) + "\n")
backup.chmod(0o600)

models = config.setdefault("models", {})
models["mode"] = "merge"
models.setdefault("providers", {})["riley-azure"] = {
    "baseUrl": "http://172.17.0.1:8787/v1",
    "apiKey": "${RILEY_PROXY_TOKEN}",
    "api": "openai-completions",
    "models": [{
        "id": deployment,
        "name": "Riley Azure Model",
        "contextWindow": 32768,
        "maxTokens": 4096
    }]
}

defaults = config.setdefault("agents", {}).setdefault("defaults", {})
defaults["model"] = {"primary": f"riley-azure/{deployment}"}

config_path.write_text(json.dumps(config, indent=2) + "\n")
config_path.chmod(0o600)

print("Riley model configured")
print("Existing tool settings preserved")
print("Backup created")

Usage: openclaw agent [options] [command]

Run an agent turn via the Gateway (use --local for embedded)

Options:

                             (omit to use the main session channel)
  --deliver                  Send the agent's reply back to the selected channel (default: false)
  -h, --help                 Display help for command
  --json                     Output result as JSON (default: false)
  --local                    Run the embedded agent locally using configured provider credentials or local CLI logins (default: false)
  -m, --message <text>       Message body for the agent
  --message-file <path>      Read the agent message body from a UTF-8 file (max 4 MiB)
  --model <id>               Model override for this run (provider/model or model id)
  --reply-account <id>       Delivery account id override
  --reply-channel <channel>  Delivery channel override (separate from routing)
  --reply-to <target>        Delivery target override (separate from session routing)
  --session-id <id>          Use an explicit session id
  --session-key <key>        Explicit session key (agent:<id>:<key>, or scoped to --agent)
  -t, --to <number>          Recipient number in E.164 used to derive the session key
  --thinking <level>         Thinking level: off | minimal | low | medium | high | xhigh | adaptive | max | ultra where supported
  --timeout <seconds>        Override agent command timeout (seconds, default 600 or config value)
  --verbose <on|off>         Persist agent verbose level for the session

Commands:
